void main(

  float4 v1 : TEXCOORD0,
  float4 v2 : TEXCOORD1,
  float4 v3 : TEXCOORD2,
  out float4 o0 : SV_Target0)
{
  float4 r0,r1,r2;
  uint4 bitmask, uiDest;
  float4 fDest;
  r0.x = dot(v3.xyz, v3.xyz);
  r0.x = rsqrt(r0.x);  
  r0.xyz = v3.xyz * r0.xxx;
  r1.xyz = -Ldynamic_pos.xyz + v2.xyz;
  r0.w = dot(r1.xyz, r1.xyz);
  r1.w = sqrt(r0.w);
  r0.w = saturate(-r0.w * Ldynamic_pos.w + 1);
  r2.xyz = Ldynamic_color.xyz * r0.www;
  r1.xyz = -r1.xyz / r1.www;
  r0.x = saturate(dot(r1.xyz, r0.xyz));
  r0.xyz = r2.xyz * r0.xxx;
  r0.w = 15 + -v2.z;
  r0.w = 0.0666666701 * r0.w;
  r1.x = r0.w * r0.w;
  r0.w = r1.x * r0.w;
  r0.w = max(0, r0.w);
  r1.x = 1 + r0.w;
  r0.w = -0.200000003 * r0.w;
  r1.y = v0.x * rtdim.x + -0.5;
  
  float4 stereo = StereoParams.Load(0);
  float depth = Ldynamic_pos.w;
  
  // Apply fix here. Also slightly moves spiderwebs, but mainly light reflection
  r1.y -= stereo.x*(1.0 - stereo.y * depth)*0.5;
  
  r1.z = rtdim.y / rtdim.x;
  r2.x = r1.y * r1.z;
  r2.yz = rtdim.xy * v0.xy;
  r1.yz = float2(0,-0.5) + r2.xz;
  r1.w = t_zbuffer.Sample(s_clamp_none_s, r2.yz).x;
  r1.w = depth_xform.x * r1.w + depth_xform.y;
  r1.w = 1 / r1.w;
  r1.w = -v2.z + r1.w;
  r1.w = saturate(r1.w + r1.w);
  r1.w = saturate(v2.w * r1.w);
  r1.xy = r1.yz * r1.xx + r0.ww;
  r1.xy = float2(0.5,0.5) + r1.xy;
  r1.xyz = t_pl_light.Sample(s_clamp_bi_s, r1.xy).xyz;
  r0.xyz = r0.xyz * r1.xyz + ambient.xyz;
  r2.xyzw = t_base.Sample(s_wrap_tri_s, v1.xy).xyzw;
  o0.xyz = r2.xyz * r0.xyz;
  r0.x = r2.w * r1.w;
  o0.w = v1.z * r0.x;
  return;
}
