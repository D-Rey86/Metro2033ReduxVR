void main(
  float4 v0 : SV_Position0,
  float4 v1 : TEXCOORD0,
  float4 v2 : TEXCOORD2,
  float4 v3 : TEXCOORD4,
  float4 v4 : TEXCOORD1,
  float2 v5 : TEXCOORD3,
  float w5 : TEXCOORD5,
  out float4 o0 : SV_Target0)
{
  float4 r0,r1,r2,r3,r4;
  uint4 bitmask, uiDest;
  float4 fDest;

  r0.xyzw = t_base.Sample(s_wrap_tri_s, v4.zw).xyzw;
  r1.xyzw = t_base.Sample(s_wrap_tri_s, v4.xy).xyzw;
  r0.xyzw = -r1.xyzw + r0.xyzw;
  r0.xyzw = v5.yyyy * r0.xyzw + r1.xyzw;
  r1.x = r0.w * v1.w + -0.00300000003;
  r0.xyzw = v1.xyzw * r0.xyzw;
  r1.x = cmp(r1.x < 0);
  if (r1.x != 0) discard;
  r1.x = v0.x * rtdim.x + -0.5;
  
  float4 stereo = StereoParams.Load(0);
  float depth = Ldynamic_pos.w;
  
  // Apply fix here. Also slightly moves spiderwebs, but mainly light reflection
  r1.x -= stereo.x*(1.0 - stereo.y * depth)*0.5;
  
  
  
  r1.y = rtdim.y / rtdim.x;
  r1.x = r1.x * r1.y;
  r1.z = rtdim.y * v0.y;
  r1.xy = float2(0,-0.5) + r1.xz;
  r1.z = 15 + -v2.z;
  r1.z = 0.0666666701 * r1.z;
  r1.w = r1.z * r1.z;
  r1.z = r1.w * r1.z;
  r1.z = max(0, r1.z);
  r1.w = 1 + r1.z;
  r1.z = -0.200000003 * r1.z;
  r1.xy = r1.xy * r1.ww + r1.zz;
  r1.xy = float2(0.5,0.5) + r1.xy;
  r1.xyz = t_pl_light.Sample(s_clamp_bi_s, r1.xy).xyz;
  r1.w = dot(v3.xyz, v3.xyz);
  r1.w = rsqrt(r1.w);
  r2.xyz = v3.xyz * r1.www;
  r3.xyz = -Ldynamic_pos.xyz + v2.xyz;
  r1.w = dot(r3.xyz, r3.xyz);
  r2.w = sqrt(r1.w);
  r1.w = saturate(-r1.w * Ldynamic_pos.w + 1);
  r4.xyz = Ldynamic_color.xyz * r1.www;
  r3.xyz = -r3.xyz / r2.www;
  r1.w = saturate(dot(r3.xyz, r2.xyz));
  r2.xyz = r4.xyz * r1.www;
  r1.xyz = r2.xyz * r1.xyz + ambient.xyz;
  o0.xyz = r1.xyz * r0.xyz;
  o0.w = saturate(r0.w);
  return;
}
