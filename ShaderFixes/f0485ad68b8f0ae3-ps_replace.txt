void main(
  float4 v0 : SV_Position0,
  float4 v1 : TEXCOORD0,
  float4 v2 : TEXCOORD2,
  float3 v3 : TEXCOORD4,
  float w3 : TEXCOORD3,
  float4 v4 : TEXCOORD1,
  float v5 : TEXCOORD5,
  out float4 o0 : SV_Target0)
{
  float4 r0,r1,r2,r3;
  uint4 bitmask, uiDest;
  float4 fDest;

  r0.xyzw = t_base.Sample(s_wrap_tri_s, v4.zw).xyzw;
  r1.xyzw = t_base.Sample(s_wrap_tri_s, v4.xy).xyzw;
  r0.xyzw = -r1.xyzw + r0.xyzw;
  r0.xyzw = w3.x * r0.xyzw + r1.xyzw;
  r1.x = r0.w * v1.w + -0.00300000003;
  r0.xyzw = v1.xyzw * r0.xyzw;
  r1.x = cmp(r1.x < 0);
  if (r1.x != 0) discard;
  r1.x = dot(v3.xyz, v3.xyz);
  r1.x = rsqrt(r1.x);  
  r1.xyz = v3.xyz * r1.xxx;
  r2.xyz = -Ldynamic_pos.xyz + v2.xyz;
  r1.w = dot(r2.xyz, r2.xyz);
  r2.w = sqrt(r1.w);
  r1.w = saturate(-r1.w * Ldynamic_pos.w + 1);
  r3.xyz = Ldynamic_color.xyz * r1.www;
  r2.xyz = -r2.xyz / r2.www;
  r1.x = saturate(dot(r2.xyz, r1.xyz));
  r1.xyz = r3.xyz * r1.xxx;
  r1.w = 15 + -v2.z;
  r1.w = 0.0666666701 * r1.w;
  r2.x = r1.w * r1.w;
  r1.w = r2.x * r1.w;
  r1.w = max(0, r1.w);
  r2.x = 1 + r1.w;
  r1.w = -0.200000003 * r1.w;
  r2.y = v0.x * rtdim.x + -0.5;
  
  float4 stereo = StereoParams.Load(0);
  float depth = Ldynamic_pos.w;
  
  // Apply fix here. Also slightly moves spiderwebs, but mainly light reflection
  r2.y -= stereo.x*(1.0 - stereo.y * depth)*0.5;
  
  r2.z = rtdim.y / rtdim.x;
  r3.x = r2.y * r2.z;
  r3.yz = rtdim.xy * v0.xy;
  r2.yz = float2(0,-0.5) + r3.xz;
  r2.w = t_zbuffer.Sample(s_clamp_none_s, r3.yz).x;
  r2.w = depth_xform.x * r2.w + depth_xform.y;
  r2.w = 1 / r2.w;
  r2.w = -v2.z + r2.w;
  r2.w = saturate(0.200000003 * r2.w);
  r2.w = saturate(v2.w * r2.w);
  o0.w = saturate(r2.w * r0.w);
  r2.xy = r2.yz * r2.xx + r1.ww;
  r2.xy = float2(0.5,0.5) + r2.xy;
  r2.xyz = t_pl_light.Sample(s_clamp_bi_s, r2.xy).xyz;
  r1.xyz = r1.xyz * r2.xyz + ambient.xyz;
  o0.xyz = r1.xyz * r0.xyz;
  return;
}
